One idea, six languages

Keep what was there before you change it, and you can always put it back exactly. It is the rule behind retractbench, written out in the languages these projects are made of.

Python where most of the work is 
GONE = object()

def edit(model, key, value, ledger):
    ledger.append((key, model.get(key, GONE)))
    model[key] = value

def undo(model, ledger):
    key, before = ledger.pop()
    if before is GONE:
        del model[key]
    else:
        model[key] = before
TypeScript StoryGecko's editor 
type Entry<K, V> = { key: K; had: boolean; before?: V };

function edit<K, V>(model: Map<K, V>, key: K, value: V, ledger: Entry<K, V>[]) {
  ledger.push({ key, had: model.has(key), before: model.get(key) });
  model.set(key, value);
}

function undo<K, V>(model: Map<K, V>, ledger: Entry<K, V>[]) {
  const last = ledger.pop();
  if (!last) return;
  if (last.had) model.set(last.key, last.before as V);
  else model.delete(last.key);
}
SQL the ledger as a table 
BEGIN;
INSERT INTO ledger (id, before)
  SELECT id, body FROM notes WHERE id = 7;
UPDATE notes SET body = 'edited' WHERE id = 7;
COMMIT;

-- and back
UPDATE notes
   SET body = (SELECT before FROM ledger WHERE ledger.id = notes.id)
 WHERE id = 7;
